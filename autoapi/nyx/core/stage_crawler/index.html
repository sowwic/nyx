    NodeVisitor ¶

    Bases: Protocol

    Visitor interface, structural rather than nominal.

    Node.accept(visitor) should call visitor.visit_node(self). Any class with a matching visit_node method satisfies this Protocol -- no inheritance required. This is the double-dispatch hook that lets new operations (crawling, validation, rendering, ...) be added over the Node hierarchy without modifying Node itself.

    runtime_checkable only lets isinstance(x, NodeVisitor) check for the presence of visit_node (not its signature) -- useful for an assertion or guard clause, not for full type safety. Static type checkers (mypy/pyright) do verify the full signature.

    StageCrawler(stage) ¶

    Post-order crawler over a Stage's node graph.

    Implements the NodeVisitor protocol structurally (via visit_node) without inheriting from it.

    Traversal rules: - A node's children are found via find_flow_start_node() and then followed via each node's core:flow components. - A node's own children (and their full subtrees) are executed before the node itself executes (post-order / bottom-up). - Branch nodes: only some of their outgoing core:flow components are is_active after execution -- resolved only after execute() runs, never guessed ahead of time. - Sequence nodes: multiple outgoing core:flow components can be is_active simultaneously -- each is traversed fully, depth-first, one after another. - core:flow components with readable == False are write/inout only and are never treated as traversable paths. - A core:flow component can only ever point to a sibling under the same parent; anything else is treated as a misconfiguration and logged. - A node that was already reached in the run is not followed again, so convergence and cycles end instead of repeating.

    The walk is depth-first but iterative: the steps still to do are kept on a stack instead of in recursive calls, so the length of a flow chain is not limited by the recursion limit. Which flow components lead on is decided by ExecutionChain, the same rule the editor uses to show the chain.

    def process_events(self, check_cancel: bool = True, force: bool = False) -> None:
    """Let the application handle events, so the UI stays alive during a run.

    Does nothing more often than every `PUMP_INTERVAL` seconds, unless forced.

    Args:
        check_cancel (bool): raise when the run was asked to stop.
        force (bool): handle events even if they were just handled.

    Raises:
        StageExecutionCancelled: the run was asked to stop.
    """
    now = time.monotonic()
    if QtCore.QCoreApplication.instance() is not None and (
        force or now - self._last_pump >= self.PUMP_INTERVAL
    ):
        self._last_pump = now
        flags = QtCore.QEventLoop.ProcessEventsFlag.AllEvents
        if not self.accept_user_input:
            flags = QtCore.QEventLoop.ProcessEventsFlag.ExcludeUserInputEvents
        QtCore.QCoreApplication.processEvents(flags)
    if check_cancel and self._cancel_requested:
        raise StageExecutionCancelled("Execution cancelled")

    def run_flow_branch(self, flow_component: FlowComponent) -> None:
    """Run the chain a flow component leads to, now, before returning.

    Meant for nodes that run a chain more than once, like a loop. The nodes of
    the chain count as not visited again afterwards, so the same chain can be
    run again.

    Args:
        flow_component (FlowComponent): flow component whose chain to run.
    """
    next_node = flow_component.get_next_node()
    if next_node is None:
        return
    if next_node.parent() != flow_component.node.parent():
        raise RuntimeError(
            f"{flow_component.path} points to "
            f"{next_node.path}, which is not a sibling node.",
        )

    visited_paths = list(self._visited_node_paths)
    visited_path_set = set(self._visited_node_path_set)
    followed_paths = set(self._followed_node_paths)
    pending_steps = self._pending_steps
    was_running_steps = self._is_running_steps
    self._pending_steps = []
    self._is_running_steps = False
    try:
        self._traverse_children(next_node)
    finally:
        self._pending_steps = pending_steps
        self._is_running_steps = was_running_steps
        self._visited_node_paths[:] = visited_paths
        self._visited_node_path_set.clear()
        self._visited_node_path_set.update(visited_path_set)
        self._followed_node_paths.clear()
        self._followed_node_paths.update(followed_paths)

    def start(self, start_node: Node | None = None) -> None:
    """Begin (or restart) a crawl from ``start_node`` or the stage root.

    A run that is asked to stop with ``request_cancel`` ends quietly, after the
    cleanups of the open scopes, with ``execution_cancelled`` emitted.
    """
    if self._is_running:
        LOGGER.warning("%s | Already running, not starting again", self.stage)
        return
    self._is_running = True
    self._cancel_requested = False
    self._last_pump = 0.0
    cancelled = False
    try:
        self.clear_cached_data()
        self._context = NodeExecutionContext(self.stage)
        self.stage.execution_started.emit()
        start_node: Node | FlowStartNode | None = start_node or self.stage.root_node
        try:
            self._traverse_children(start_node)  # kicks off the descent
        except StageExecutionCancelled:
            cancelled = True
        finally:
            # A failing node leaves scopes open, their cleanups still have to run.
            self._context.unwind()
    finally:
        self._is_running = False
        self.accept_user_input = False
    if cancelled:
        LOGGER.info("Execution cancelled")
        self.stage.execution_cancelled.emit()
        return
    self.stage.execution_succeeded.emit()
    self.stage.execution_finished.emit()

    def visit_node(self, node: Node) -> None:
    """Visit a node, executing it and its children in the type's order.

    Nodes already visited in this run are skipped.

    Args:
        node (Node): node to visit.
    """
    node_path_as_string = str(node.path)

    if self._has_node_already_been_visited(node_path_as_string):
        LOGGER.warning(
            "Cycle/convergence at %s, not re-visiting", node_path_as_string
        )
        return
    self._visited_node_paths.append(node_path_as_string)
    self._visited_node_path_set.add(node_path_as_string)

    node_execution_order = node.get_type_execution_order()

    # The scope of a node lasts while its children execute. It is closed by a
    # step scheduled before them, which runs once all of them are done.
    if node_execution_order is NodeExecutionOrderFlag.BEFORE_CHILDREN:
        self._schedule(
            functools.partial(self._execute_node_and_record_its_path, node),
            functools.partial(self._open_scope_of_node, node),
            functools.partial(self._resolve_children_of_node, node),
            self._context_pop_scope,
        )
    elif node_execution_order is NodeExecutionOrderFlag.AFTER_CHILDREN:
        self._schedule(
            functools.partial(self._open_scope_of_node, node),
            functools.partial(self._resolve_children_of_node, node),
            self._context_pop_scope,
            functools.partial(self._execute_node_and_record_its_path, node),
        )
    self._run_scheduled_steps()

    def wait_for(self, func: collections.abc.Callable, *args, **kwargs):
    """Call a blocking function without freezing the UI while it blocks.

    The function runs in another thread, so it must be safe to call from one. If
    the run is cancelled meanwhile, it is left to finish and its result dropped.

    Args:
        func (Callable): function to call.
        *args: positional arguments of the function.
        **kwargs: keyword arguments of the function.

    Returns:
        Any: what the function returned.

    Raises:
        StageExecutionCancelled: the run was asked to stop.
    """
    outcome: dict[str, typing.Any] = {}

    def call() -> None:
        try:
            outcome["value"] = func(*args, **kwargs)
        except BaseException as err:  # re-raised in the calling thread
            outcome["error"] = err

    thread = threading.Thread(target=call, name="StageCrawler.wait_for", daemon=True)
    thread.start()
    while thread.is_alive():
        thread.join(self.PUMP_INTERVAL)
        self.process_events()
    if "error" in outcome:
        raise outcome["error"]
    return outcome.get("value")
